+function Get-MavenDependencies {

+            ArtifactId = $resolvedArtifactId
+            Scope = if ([string]::IsNullOrWhiteSpace($scope)) { "compile" } else { Resolve-MavenPropertyValue -Value $scope -Properties $properties }
+            Optional = [string]::Equals((Resolve-MavenPropertyValue -Value $optional -Properties $properties), "true", [System.StringComparison]::OrdinalIgnoreCase)

+function Resolve-MavenPropertyValue {
+    param(
+        [string] $Value,
+        [hashtable] $Properties
+    )
+
+    if ([string]::IsNullOrWhiteSpace($Value)) {
+        return $Value
+    }
+
+    $resolvedValue = $Value.Trim()
+    for ($i = 0; $i -lt 10; $i++) {
+        $changed = $false
+        foreach ($match in [regex]::Matches($resolvedValue, '\$\{([^}]+)\}')) {
+            $propertyName = $match.Groups[1].Value
+            if ($Properties.ContainsKey($propertyName)) {
+                $resolvedValue = $resolvedValue.Replace($match.Value, [string] $Properties[$propertyName])
+                $changed = $true
+            }
+        }
+        if (-not $changed) {
+            break
+        }
+    }
+    return $resolvedValue
+}
